cmake_minimum_required(VERSION 3.16)
project(CoconutQtTheme LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(COCONUT_QT_MAJOR 6 CACHE STRING "Qt major version to build against")
find_package(Qt${COCONUT_QT_MAJOR} REQUIRED COMPONENTS Core Gui Widgets DBus)
if(COCONUT_QT_MAJOR EQUAL 6)
    find_package(Qt6GuiPrivate REQUIRED)
endif()
add_library(coconut-qt${COCONUT_QT_MAJOR} MODULE coconut.cpp)
target_link_libraries(coconut-qt${COCONUT_QT_MAJOR} PRIVATE Qt${COCONUT_QT_MAJOR}::Core Qt${COCONUT_QT_MAJOR}::Gui Qt${COCONUT_QT_MAJOR}::GuiPrivate Qt${COCONUT_QT_MAJOR}::Widgets Qt${COCONUT_QT_MAJOR}::DBus)
set_target_properties(coconut-qt${COCONUT_QT_MAJOR} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/platformthemes")
install(TARGETS coconut-qt${COCONUT_QT_MAJOR} LIBRARY DESTINATION "share/coconut/qt/plugins/platformthemes")
include(CTest)
if(BUILD_TESTING)
    add_executable(coconut-qt-probe probe.cpp)
    target_link_libraries(coconut-qt-probe PRIVATE Qt${COCONUT_QT_MAJOR}::Core Qt${COCONUT_QT_MAJOR}::Gui Qt${COCONUT_QT_MAJOR}::Widgets)
endif()
